{{/* Open Graph and Twitter Card tags, so a link pasted into LinkedIn, Slack or Mastodon renders a card with the hero image rather than bare title-and-domain. og:image has to be an absolute URL. Hero images already are (Unsplash), and the w/h are rewritten from the 800x400 the page renders to the 1200x630 the social scrapers want. */}} {{- $title := cond .IsHome .Site.Title (printf "%s · %s" .Title .Site.Title) -}} {{- $desc := cond .IsHome .Site.Params.description (.Summary | plainify | truncate 200) -}} {{- if not $desc }}{{ $desc = .Site.Params.description }}{{ end -}} {{- $image := "" -}} {{- with .Params.hero_image -}} {{- $image = replaceRE `w=\d+&h=\d+` "w=1200&h=630" . -}} {{- else -}} {{- with .Site.Params.ogImage -}}{{- $image = absURL . -}}{{- end -}} {{- end }} {{- with $image }} {{- end }} {{- if eq .Section "posts" }} {{- range .Params.tags }} {{- end }} {{- end }} {{- with $image }} {{- end }}
